 the loop that draws every petal
for ring in rings:
    for i in range(ring["count"]):
        angle = (360 / ring["count"]) * i + ring["offset"]
        base_x = ring["radius"] * cos(radians(angle))
        base_y = ring["radius"] * sin(radians(angle))
        draw_petal(base=(base_x, base_y), base_angle=angle,
                   length=ring["length"], width=ring["width"],
                   color=ring["color"])  # pen never lifts
